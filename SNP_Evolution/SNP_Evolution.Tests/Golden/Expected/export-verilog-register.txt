// register: 5 neurons, exported from SNP_Evolution. One clock is one step of the SN P system.
// Each step every neuron applies its one applicable rule, if any, then the spikes sent this step are delivered
// to every open target along with the environment's spikes; a closed neuron loses what is sent to it.

module register (
    input  wire clk,
    input  wire rst,    // synchronous: loads every neuron's initial spikes
    input  wire [4:0] start,    // spikes arriving at n1 this step
    input  wire [4:0] n,    // spikes arriving at n2 this step
    output wire [2:0] out,    // spikes n3 sends this step
    output wire [2:0] done,    // spikes n4 sends this step
    output wire overflow    // set once a counter would need more bits than it has
);
    wire [2:0] sent_1;
    wire [2:0] sent_2;
    wire [2:0] sent_3;
    wire [2:0] sent_4;
    wire [2:0] sent_5;
    wire [4:0] overflows;

    wire [4:0] incoming_1 = start;
    register_n1 n1 (.clk(clk), .rst(rst), .incoming(incoming_1), .sent(sent_1), .count(), .overflow(overflows[0]));
    wire [4:0] incoming_2 = n;
    register_n2 n2 (.clk(clk), .rst(rst), .incoming(incoming_2), .sent(sent_2), .count(), .overflow(overflows[1]));
    wire [2:0] incoming_3 = sent_5;
    register_n3 n3 (.clk(clk), .rst(rst), .incoming(incoming_3), .sent(sent_3), .count(), .overflow(overflows[2]));
    wire [2:0] incoming_4 = sent_5;
    register_n4 n4 (.clk(clk), .rst(rst), .incoming(incoming_4), .sent(sent_4), .count(), .overflow(overflows[3]));
    wire [3:0] incoming_5 = sent_1 + sent_2;
    register_n5 n5 (.clk(clk), .rst(rst), .incoming(incoming_5), .sent(sent_5), .count(), .overflow(overflows[4]));

    assign out = sent_3;
    assign done = sent_4;
    assign overflow = |overflows;
endmodule

// n1: a -> a; sends to n5
module register_n1 (
    input  wire clk,
    input  wire rst,
    input  wire [4:0] incoming,
    output reg  [2:0] sent,
    output reg  [4:0] count,
    output reg  overflow
);
    localparam INITIAL = 0;
    localparam [5:0] MOST = 31;
    // Rule 1, a -> a: bit i of ACCEPTS says whether lasso position i matches,
    // and a count past the tail of 2 goes round the period of 1.
    localparam R1_TAIL = 2;
    localparam R1_PERIOD = 1;
    localparam [2:0] R1_ACCEPTS = 3'b010;

    reg [4:0] next_count;
    reg [2:0] own;
    reg [5:0] total;
    wire r1 = count >= 1 && ((count < R1_TAIL) ? R1_ACCEPTS[count] : R1_ACCEPTS[R1_TAIL + (count - R1_TAIL) % R1_PERIOD]);

    always @* begin
        next_count = count;
        own = 0;
        if (r1) begin
            own = 1;
            next_count = count - 1;
        end
        sent = own;
        total = next_count + incoming;
    end

    always @(posedge clk) begin
        if (rst) begin
            count <= INITIAL;
            overflow <= 0;
        end else begin
            count <= total[4:0];
            overflow <= overflow || total > MOST;
        end
    end
endmodule

    wire [2:0] out;

    reg [2:0] sent_2;
    reg [2:0] sent_3;
    reg [2:0] sent_4;
    reg [2:0] sent_5;
    register dut (.clk(clk), .rst(rst), .start(start), .n(n), .out(out), .done(done), .overflow(overflow));

        begin
            #1 clk = 1;
            #1 clk = 0;
        end
    endtask

    task step;
        begin
            #1;
            sent_1 = dut.n1.sent;
            sent_2 = dut.n2.sent;
            sent_3 = dut.n3.sent;
            sent_4 = dut.n4.sent;
            sent_5 = dut.n5.sent;

            $display("%0d 1 %0d %0d", t, sent_1, dut.n1.count);
            $display("%0d 2 %0d %0d", t, sent_2, dut.n2.count);
            $display("%0d 3 %0d %0d", t, sent_3, dut.n3.count);
            $display("%0d 4 %0d %0d", t, sent_4, dut.n4.count);
            $display("%0d 5 %0d %0d", t, sent_5, dut.n5.count);
        end
    endtask
