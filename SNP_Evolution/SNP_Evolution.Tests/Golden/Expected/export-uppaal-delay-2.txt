void land(int i) {
  int arriving = flight[i][step % SLOTS];
  if (arriving &gt; 0) { flight[i][step % SLOTS] = 0; emit[i] += arriving; }
}
void busy(int i) {
  emit[i] = 0;
  closedNow[i] = false;
  closedFor[i]--;
  if (closedFor[i] &gt; 0) { closedNow[i] = true; }
  else { emit[i] = pending[i]; pending[i] = 0; }
  land(i);
}
void idle(int i) {
  emit[i] = 0;
  closedNow[i] = false;
  land(i);
}
void fireRule(int i, int consume, int produce, int delay, bool axonal) {
  emit[i] = 0;
  closedNow[i] = false;
  if (consume &lt; 0) spikes[i] = 0; else spikes[i] -= consume;
  if (axonal) { flight[i][(step + delay) % SLOTS] += produce; }
  else if (delay &gt; 0) { closedFor[i] = delay; pending[i] = produce; closedNow[i] = true; }
  else { emit[i] = produce; }
  land(i);
}

// The verdicts, once the run is over, as ContractTask scores a run.
void judge() {
  int i;
  backOk = true;
  for (i = 0; i &lt; NEURONS; i++) { if (spikes[i] != INITIAL[i]) backOk = false; }
  onTimeOk = firstDone &gt;= 0 &amp;&amp; firstDone - (START[caseNo] + 1) &gt;= MIN_LATENCY &amp;&amp; firstDone - (START[caseNo] + 1) &lt;= MAX_LATENCY;
  doneOk = doneFirings == 1 &amp;&amp; doneSlot == EXPECTED_DONE[caseNo];
}

void deliver() {
  int i;
  if (emit[0] &gt; 0) { if (!closedNow[1]) spikes[1] += emit[0]; }
  if (caseNo == 0 &amp;&amp; step == 2 &amp;&amp; !closedNow[0]) spikes[0] += 1;
  if (step &lt;= START[caseNo] &amp;&amp; (emit[1] &gt; 0)) quietBroken = true;
  if (emit[1] &gt; 0) { doneFirings++; if (doneSlot &lt; 0) doneSlot = 0; if (firstDone &lt; 0) firstDone = step; }
  for (i = 0; i &lt; NEURONS; i++) { if (spikes[i] != EXPECT[caseNo][step][i]) traceBroken = true; }
  step++;
  if ((firstDone &gt;= 0 &amp;&amp; step &gt; firstDone + AFTER_DONE) || step == HORIZON) { ended = true; judge(); }
}
